local function httpRequest(url, method, headers)
    local rf = (syn and syn.request) or (http and http.request) or http_request or request
    if rf then
        local res = rf({ Url = url, Method = method or "GET", Headers = headers or {} })
        if type(res) == "table" then return res.Body else return res end
    else
        return game:HttpGet(url)
    end
end

local function copyToClipboard(text)
    if setclipboard then setclipboard(text)
    elseif toclipboard then toclipboard(text) end
end

local function verifyModkey(inputKey)
    local endpoint = string.format("%s/softs/%s/keys/%s", BASE_URL, SOFT_ID, HttpService:UrlEncode(inputKey))
    local headers = { ["Authorization"] = "Bearer " .. API_KEY, ["Content-Type"] = "application/json" }
    local success, body = pcall(function() return httpRequest(endpoint, "GET", headers) end)
    if success and body then
        local ok, res = pcall(function() return HttpService:JSONDecode(body) end)
        if ok and res and res.ok == true and res.data then
            local data = res.data
            if data.active == 0 then return false, "Key Expired" end
            if data.status == 0 then return false, "Key Disabled" end
            local rawExp = data.expired_date or data.expires_at
            if rawExp then
                local num = tonumber(rawExp)
                if num then
                    if num > 20000000000 then num = math.floor(num / 1000) end
                    if num > 0 and os.time() > num then return false, "Key Expired" end
                end
            end
            return true, "License Valid", data
        else
            return false, res and res.message or "Invalid Key"
        end
    end
    return false, "Connection Failed"
end

local function loadMainScript()
    print("Loading Main Script...")
    local success, err = pcall(function()
        local script = game:HttpGet(MAIN_SCRIPT_URL)
        if script then
            loadstring(script)()
        end
    end)
    if not success then
        warn("Main Script Error: " .. tostring(err))
        status.Text = "Error loading main script!"
        status.TextColor3 = Color3.fromRGB(255, 80, 80)
    end
end
