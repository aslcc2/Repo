local function httpRequest(url, method, headers)
    local requestFunc = (syn and syn.request) or (http and http.request) or http_request or request or (fluxus and fluxus.request)
    if requestFunc then
        local response = requestFunc({
            Url = url,
            Method = method or "GET",
            Headers = headers or {}
        })
        if type(response) == "table" then return response.Body else return response end
    else
        return game:HttpGet(url)
    end
end

local function parseExpDate(data)
    if type(data) ~= "table" then return "Lifetime" end
    local rawExp = data.expired_date or data.expires_at or data.expired_at or data.expire_date or data.expiration or data.expired
    
    if not rawExp or rawExp == "" or rawExp == 0 then return "Lifetime" end

    local expStr = tostring(rawExp)
    if string.lower(expStr) == "lifetime" or string.lower(expStr) == "never" or rawExp == -1 then
        return "Lifetime"
    end

    local numTime = tonumber(rawExp)
    if numTime then
        if numTime > 20000000000 then numTime = math.floor(numTime / 1000) end
        if numTime > 0 then return os.date("%Y-%m-%d %H:%M", numTime) end
    end

    local year, month, day, hour, min = string.match(expStr, "(%d%d%d%d)[%-%/](%d%d)[%-%/](%d%d)[%sT]?(%d?%d?):?(%d?%d?)")
    if year and month and day then
        if hour ~= "" and min ~= "" then
            return string.format("%s-%s-%s %s:%s", year, month, day, hour, min)
        end
        return string.format("%s-%s-%s", year, month, day)
    end

    return string.sub(expStr, 1, 16)
end

local function parseMaxDevices(data)
    if type(data) ~= "table" then return "1" end
    local dev = data.max_devices or data.max_device or data.device_limit or data.devices_limit or data.limit or data.max_hwid
    return dev and tostring(dev) or "1"
end

local function verifyModkey(inputKey)
    local endpoint = string.format("%s/softs/%s/keys/%s", BASE_URL, SOFT_ID, HttpService:UrlEncode(inputKey))
    local headers = {
        ["Authorization"] = "Bearer " .. API_KEY,
        ["Content-Type"] = "application/json"
    }

    local success, body = pcall(function() return httpRequest(endpoint, "GET", headers) end)

    if success and body then
        local decodeSuccess, response = pcall(function() return HttpService:JSONDecode(body) end)
        if decodeSuccess and response then
            if response.ok == true and response.data then
                local data = response.data
                if data.active == 0 then return false, "Key Expired / Paused!", data
                elseif data.status == 0 then return false, "Key Disabled!", data end

                local rawExp = data.expired_date or data.expires_at or data.expired_at or data.expire_date
                if rawExp then
                    local numTime = tonumber(rawExp)
                    if numTime then
                        if numTime > 20000000000 then numTime = math.floor(numTime / 1000) end
                        if numTime > 0 and os.time() > numTime then
                            return false, "Key Has Expired!", data
                        end
                    end
                end

                return true, "License Valid!", data
            else
                local msg = response.message or "Invalid License Key!"
                if string.find(string.lower(msg), "expire") then msg = "Key Has Expired!" end
                return false, msg, nil
            end
        end
    end
    return false, "Connection Failed!", nil
end

local nodes = {}
local RADIUS = 110
local LAT_COUNT = 10
local LON_COUNT = 16

for i = 1, LAT_COUNT do
    local lat = math.rad(-90 + (i * 180 / (LAT_COUNT + 1)))
    for j = 1, LON_COUNT do
        local lon = math.rad(j * 360 / LON_COUNT)
        local x = RADIUS * math.cos(lat) * math.cos(lon)
        local y = RADIUS * math.sin(lat)
        local z = RADIUS * math.cos(lat) * math.sin(lon)
        
        local p = Instance.new("Frame")
        p.Size = UDim2.new(0, 3, 0, 3)
        p.AnchorPoint = Vector2.new(0.5, 0.5)
        p.BackgroundColor3 = isTWDGame and Color3.fromRGB(0, 240, 255) or Color3.fromRGB(255, 50, 75)

        p.Parent = globeFrame
        
        local pCorner = Instance.new("UICorner")
        pCorner.CornerRadius = UDim.new(1, 0)
        pCorner.Parent = p
        
        table.insert(nodes, { Frame = p, BasePos = Vector3.new(x, y, z) })
    end
end

    orbitRing.Rotation = orbitRing.Rotation + (dt * 15)
    
    local cosY = math.cos(angleY)
    local sinY = math.sin(angleY)
    
    for _, node in ipairs(nodes) do
        local v = node.BasePos
        local rx = v.X * cosY - v.Z * sinY
        local ry = v.Y
        local rz = v.X * sinY + v.Z * cosY
        
        local tilt = math.rad(18)
        local finalY = ry * math.cos(tilt) - rz * math.sin(tilt)
        local finalZ = ry * math.sin(tilt) + rz * math.cos(tilt)
        
        local scale = 300 / (300 + finalZ)
        local screenX = rx * scale
        local screenY = finalY * scale
        
        node.Frame.Position = UDim2.new(0.5, screenX, 0.5, screenY)
        local depthAlpha = (finalZ + RADIUS) / (2 * RADIUS)
        node.Frame.BackgroundTransparency = math.clamp(0.1 + (depthAlpha * 0.8), 0.1, 0.95)
        
        if isTWDGame then
            if finalZ < 0 then node.Frame.BackgroundColor3 = Color3.fromRGB(0, 255, 230)
            else node.Frame.BackgroundColor3 = Color3.fromRGB(130, 0, 255) end
        else
            node.Frame.BackgroundColor3 = Color3.fromRGB(255, 50, 75)
        end
    end
end)

    local function isSameTeam(plr)
        local teamCheckActive = false
        if settings.teamCheckESP or settings.aimbotTeamCheck or settings.wallhackTeamCheck or settings.hitboxTeamCheck then
            teamCheckActive = true
        end
        if not teamCheckActive then return false end
        
        local myTeam = LocalPlayer:GetAttribute("Team") or 
                       LocalPlayer:GetAttribute("Clan") or 
                       LocalPlayer:GetAttribute("Faction") or 
                       LocalPlayer:GetAttribute("Group")
        
        local theirTeam = plr:GetAttribute("Team") or 
                          plr:GetAttribute("Clan") or 
                          plr:GetAttribute("Faction") or 
                          plr:GetAttribute("Group")
        
        if myTeam and theirTeam then
            if type(myTeam) == "string" and type(theirTeam) == "string" then
                if myTeam == theirTeam then return true end
            elseif type(myTeam) == "number" and type(theirTeam) == "number" then
                if myTeam == theirTeam then return true end
            end
        end
        
        if LocalPlayer.Team and plr.Team then
            if LocalPlayer.Team == plr.Team then return true end
        end
        
        local myColor = LocalPlayer:GetAttribute("TeamColor")
        local theirColor = plr:GetAttribute("TeamColor")
        if myColor and theirColor then
            if myColor == theirColor then return true end
        end
        
        if LocalPlayer:GetAttribute("Neutral") == true and plr:GetAttribute("Neutral") == true then
            return true
        end
        
        local myLS = LocalPlayer:FindFirstChild("leaderstats")
        local theirLS = plr:FindFirstChild("leaderstats")
        if myLS and theirLS then
            local myTeamStat = myLS:FindFirstChild("Team") or myLS:FindFirstChild("Clan") or myLS:FindFirstChild("Faction")
            local theirTeamStat = theirLS:FindFirstChild("Team") or theirLS:FindFirstChild("Clan") or theirLS:FindFirstChild("Faction")
            if myTeamStat and theirTeamStat then
                if type(myTeamStat.Value) == "string" and type(theirTeamStat.Value) == "string" then
                    if myTeamStat.Value == theirTeamStat.Value then return true end
                elseif type(myTeamStat.Value) == "number" and type(theirTeamStat.Value) == "number" then
                    if myTeamStat.Value == theirTeamStat.Value then return true end
                end
            end
        end
        
        if plr:GetAttribute("Friendly") == true or plr:GetAttribute("Enemy") == false then
            return true
        end
        
        return false
    end

    local function isVisible(part)
        if not settings.visibleCheck or not part then return true end
        local origin = Camera.CFrame.Position
        local direction = (part.Position - origin).Unit * (part.Position - origin).Magnitude
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Blacklist
        params.FilterDescendantsInstances = {LocalPlayer.Character}
        local hit = Workspace:Raycast(origin, direction, params)
        if hit and hit.Instance then
            local distToHit = (hit.Position - origin).Magnitude
            local distToTarget = (part.Position - origin).Magnitude
            return distToHit >= distToTarget - 1
        end
        return true
    end

    local function hopToPlayerServer(playerName)
        if not playerName or playerName == "" then
            print("Please enter a player name")
            return
        end
        
        print("Searching for server containing player: " .. playerName)
        local servers = {}
        
        local success, result = pcall(function()
            return HttpService:JSONDecode(HttpService:GetAsync("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?limit=100"))
        end)
        
        if not success or not result or not result.data then
            print("Failed to fetch server list")
            return
        end
        
        for _, server in ipairs(result.data) do
            if server.playing ~= server.maxPlayers then
                table.insert(servers, {
                    id = server.id,
                    playing = server.playing,
                    maxPlayers = server.maxPlayers,
                    ping = server.ping
                })
            end
        end
        
        if #servers == 0 then
            print("No available servers found")
            return
        end
        
        local targetServer = nil
        for _, s in ipairs(servers) do
            local playerSuccess, playerResult = pcall(function()
                return HttpService:JSONDecode(HttpService:GetAsync("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/" .. s.id .. "/players?limit=100"))
            end)
            if playerSuccess and playerResult and playerResult.data then
                for _, p in ipairs(playerResult.data) do
                    if string.find(string.lower(p.name), string.lower(playerName)) then
                        targetServer = s
                        print("Found player " .. p.name .. " in server " .. s.id)
                        break
                    end
                end
            end
            if targetServer then break end
            task.wait(0.1)
        end
        
        if targetServer then
            print("Hoping to server: " .. targetServer.id)
            TeleportService:TeleportToInstance(game.PlaceId, targetServer.id, LocalPlayer)
        else
            print("Player not found, hopping to best available...")
            hopServerAuto()
        end
    end

    local function hopServerAuto()
        print("Searching for best server...")
        local servers = {}
        
        local success, result = pcall(function()
            return HttpService:JSONDecode(HttpService:GetAsync("https://games.roblox.com/v1/games/" .. game.PlaceId .. "/servers/Public?limit=100"))
        end)
        
        if success and result and result.data then

                        id = server.id,
                        playing = server.playing,
                        maxPlayers = server.maxPlayers,
                        ping = server.ping
                    })
                end
            end
        end
        
        if #servers > 0 then
            table.sort(servers, function(a, b) return a.playing < b.playing end)
            local target = servers[1]
            print("Hoping to server: " .. target.id)
            TeleportService:TeleportToInstance(game.PlaceId, target.id, LocalPlayer)
        else
            print("No available servers, rejoining...")
            rejoinServer()
        end
    end

    local function createWallhack(char)
        if not char or not char:FindFirstChild("HumanoidRootPart") then return end
        if char:FindFirstChild("ClientGlow") then return end
        
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum or hum.Health <= 0 then return end
        
        local player = Players:GetPlayerFromCharacter(char)
        if not player then return end
        
        if settings.wallhackTeamCheck and isSameTeam(player) then return end
        
        local highlight = Instance.new("Highlight")
        highlight.Name = "ClientGlow"
        highlight.Adornee = char
        highlight.FillColor = settings.wallhackColorVisible
        highlight.OutlineColor = settings.wallhackColorVisible
        highlight.FillTransparency = 0.4
        highlight.OutlineTransparency = 0.1
        highlight.DepthMode = Enum.HighlightDepthMode.AlwaysOnTop
        highlight.Parent = char
        
        wallhackHighlights[char] = highlight
        return highlight
    end

    local function removeWallhack(char)
        if wallhackHighlights[char] then
            pcall(function() wallhackHighlights[char]:Destroy() end)

    local function updateWallhack()
        if not settings.wallhackEnabled then
            removeAllWallhack()
            return
        end
        
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer then continue end
            
            local char = player.Character
            if not char then
                removeWallhack(char)
                continue
            end
            
            local hum = char:FindFirstChildOfClass("Humanoid")
            if not hum or hum.Health <= 0 then
                removeWallhack(char)
                continue
            end
            
            if settings.wallhackTeamCheck and isSameTeam(player) then
                removeWallhack(char)
                continue
            end
            
            local root = char:FindFirstChild("HumanoidRootPart")
            if not root then continue end
            
            local visible = true
            if settings.wallhackVisibleCheck then
                visible = isVisible(root)
            end
            
            local color = visible and settings.wallhackColorVisible or settings.wallhackColorWall
            
            local hl = wallhackHighlights[char]
            if not hl then
                createWallhack(char)
            else
                hl.FillColor = color
                hl.OutlineColor = color
            end
        end
    end

    -- ============================================================
    -- HITBOX
    -- ============================================================
    local hitboxOriginal = {}
    local hitboxBodyParts = {"UpperTorso", "Head", "HumanoidRootPart"}

    local function saveHitboxPart(player, part)
        if not hitboxOriginal[player] then hitboxOriginal[player] = {} end
        if not hitboxOriginal[player][part.Name] then
            hitboxOriginal[player][part.Name] = {
                CanCollide = part.CanCollide,
                Transparency = part.Transparency,
                Size = part.Size
            }
        end
    end

    local function restoreHitboxPart(player)
        if hitboxOriginal[player] then
            for name, props in pairs(hitboxOriginal[player]) do
                local part = player.Character and player.Character:FindFirstChild(name)
                if part and part:IsA("BasePart") then
                    part.CanCollide = props.CanCollide
                    part.Transparency = props.Transparency
                    part.Size = props.Size
                end
            end
        end
    end

    local function expandHitbox(player)
        if settings.hitboxTeamCheck and isSameTeam(player) then
            restoreHitboxPart(player)
            return
        end
        
        for _, partName in ipairs(hitboxBodyParts) do
            local part = player.Character and player.Character:FindFirstChild(partName)
            if part and part:IsA("BasePart") then
                saveHitboxPart(player, part)
                part.CanCollide = not settings.hitboxNoCollision
                part.Transparency = settings.hitboxTransparency / 10
                part.Size = Vector3.new(settings.hitboxSize, settings.hitboxSize, settings.hitboxSize)
            end
        end
    end

    local function updateHitbox()
        if not settings.hitboxEnabled then
            for player, _ in pairs(hitboxOriginal) do
                restoreHitboxPart(player)
            end
            hitboxOriginal = {}
            return
        end
        
        for _, player in ipairs(Players:GetPlayers()) do
            if player == LocalPlayer then continue end
            if not player.Character or not player.Character:FindFirstChild("HumanoidRootPart") then continue end
            expandHitbox(player)
        end
    end

    local function getPlayerWeapon(char)
        if not char then return nil end
        local tool = char:FindFirstChildOfClass("Tool")
        if tool then return tool.Name end
        for _, child in ipairs(char:GetChildren()) do
            if child:IsA("Tool") or child:IsA("Model") then
                local name = child.Name
                local weaponKeywords = {"Weapon","Gun","Sword","Axe","Knife","Machete","Katana","Pistol","Shotgun","Rifle","Bow","Hammer","Crowbar","Sniper","RPG","Launcher","SMG","Assault","Carbine","Revolver","Crossbow","Spear","Scythe","Claymore","Dagger","Fist","Claw"}
                for _, kw in ipairs(weaponKeywords) do
                    if name:find(kw) then return name end
                end
            end
        end
        local weaponAttr = char:GetAttribute("Weapon") or char:GetAttribute("Equipped") or char:GetAttribute("CurrentWeapon")
        if weaponAttr then return tostring(weaponAttr) end
        return nil
    end

    -- CAR TRACER
    local function UpdateCarTracers()
        for _, line in pairs(CarTracerLines) do
            pcall(function() line:Destroy() end)
        end
        CarTracerLines = {}
        
        if not settings.carTracer or not CarESPEnabled then
            return
        end
        
        local viewport = Camera.ViewportSize
        for Object, Data in pairs(CarESPObjects) do
            if Object and Object.Parent then
                local screenPos, onScreen = Camera:WorldToViewportPoint(Object.Position)
                if onScreen then
                    local line = Instance.new("Frame")
                    line.BorderSizePixel = 0
                    line.BackgroundColor3 = settings.carColor or Color3.fromRGB(255, 200, 0)
                    line.AnchorPoint = Vector2.new(0.5, 0.5)
                    line.BackgroundTransparency = 0.3
                    
                    local from = Vector2.new(viewport.X / 2, viewport.Y)
                    local to = Vector2.new(screenPos.X, screenPos.Y)
                    local dist = (to - from).Magnitude
                    local mid = (from + to) / 2
                    
                    if dist > 5 then
                        line.Position = UDim2.new(0, mid.X, 0, mid.Y)
                        line.Size = UDim2.new(0, dist, 0, 1.5)
                        line.Rotation = math.deg(math.atan2(to.Y - from.Y, to.X - from.X))
                        line.Parent = espScreenGui
                        table.insert(CarTracerLines, line)
                    end
                end
            end
        end
    end

            if CarESPEnabled then

                if carScanTimer >= 2 then

                end

                local count = 0

                for _, plr in ipairs(Players:GetPlayers()) do
                    if plr == LocalPlayer then continue end
                    if settings.teamCheckESP and isSameTeam(plr) then continue end
                    local char = plr.Character
                    if not char then continue end
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if not hum or hum.Health <= 0 then continue end
                    local root = char:FindFirstChild("HumanoidRootPart")
                    if not root then continue end
                    local dist = (cameraPos - root.Position).Magnitude
                    if dist > enemyMaxDist then continue end
                    count = count + 1
                end
                playerCount = count
            end
            
            if not settings.espMaster then
                hideAllESP()

                    countLabel.Text = "P:" .. playerCount
                    countFrame.Visible = true
                else
                    countFrame.Visible = false
                end
                return
            end
            
            for _, plr in ipairs(Players:GetPlayers()) do
                if plr == LocalPlayer then 
                    local data = ESPObjects[plr]

                        for _, line in pairs(data.Box) do line.Visible = false end

                        if data.HealthText then data.HealthText.Visible = false end
                        if data.HealthBarBG then data.HealthBarBG.Visible = false end
                        if data.HealthBarFG then data.HealthBarFG.Visible = false end
                        if data.DistText then data.DistText.Visible = false end
                    end
                    continue 
                end
                
                local char = plr.Character
                if not char then
                    local data = ESPObjects[plr]
                    if data then
                        for _, line in pairs(data.Box) do line.Visible = false end
                        if data.HeadCircle then data.HeadCircle.Visible = false end
                        if data.Name then data.Name.Visible = false end
                        if data.HealthText then data.HealthText.Visible = false end
                        if data.HealthBarBG then data.HealthBarBG.Visible = false end
                        if data.HealthBarFG then data.HealthBarFG.Visible = false end
                        if data.DistText then data.DistText.Visible = false end
                    end
                    continue
                end
                
                local hum = char:FindFirstChildOfClass("Humanoid")
                local root = char:FindFirstChild("HumanoidRootPart")
                local head = char:FindFirstChild("Head")
                
                if not (root and head) then
                    local data = ESPObjects[plr]
                    if data then
                        for _, line in pairs(data.Box) do line.Visible = false end
                        if data.HeadCircle then data.HeadCircle.Visible = false end
                        if data.Name then data.Name.Visible = false end
                        if data.HealthText then data.HealthText.Visible = false end
                        if data.HealthBarBG then data.HealthBarBG.Visible = false end
                        if data.HealthBarFG then data.HealthBarFG.Visible = false end
                        if data.DistText then data.DistText.Visible = false end
                    end
                    continue
                end
                
                if hum and hum.Health <= 0 then
                    local data = ESPObjects[plr]
                    if data then
                        for _, line in pairs(data.Box) do line.Visible = false end
                        if data.HeadCircle then data.HeadCircle.Visible = false end
                        if data.Name then data.Name.Visible = false end
                        if data.HealthText then data.HealthText.Visible = false end
                        if data.HealthBarBG then data.HealthBarBG.Visible = false end
                        if data.HealthBarFG then data.HealthBarFG.Visible = false end
                        if data.DistText then data.DistText.Visible = false end
                    end
                    continue
                end
                
                if settings.teamCheckESP then

                    if sameTeam then
                        local data = ESPObjects[plr]
                        if data then
                            for _, line in pairs(data.Box) do line.Visible = false end
                            if data.HeadCircle then data.HeadCircle.Visible = false end
                            if data.Name then data.Name.Visible = false end
                            if data.HealthText then data.HealthText.Visible = false end
                            if data.HealthBarBG then data.HealthBarBG.Visible = false end
                            if data.HealthBarFG then data.HealthBarFG.Visible = false end
                            if data.DistText then data.DistText.Visible = false end

                    end
                end

                local distance = (cameraPos - root.Position).Magnitude
                if distance > settings.espMaxDist then
                    local data = ESPObjects[plr]
                    if data then
                        for _, line in pairs(data.Box) do line.Visible = false end
                        if data.HeadCircle then data.HeadCircle.Visible = false end
                        if data.Name then data.Name.Visible = false end
                        if data.HealthText then data.HealthText.Visible = false end
                        if data.HealthBarBG then data.HealthBarBG.Visible = false end
                        if data.HealthBarFG then data.HealthBarFG.Visible = false end
                        if data.DistText then data.DistText.Visible = false end
                    end
                    continue
                end

                local visible = isVisible(root)
                local color
                if settings.visibleColorChange then
                    color = visible and espColor or wallColor
                else
                    color = espColor
                end

                local headScreen, headVisible = Camera:WorldToViewportPoint(head.Position)
                if not headVisible then
                    local data = ESPObjects[plr]
                    if data then
                        for _, line in pairs(data.Box) do line.Visible = false end
                        if data.HeadCircle then data.HeadCircle.Visible = false end
                        if data.Name then data.Name.Visible = false end
                        if data.HealthText then data.HealthText.Visible = false end
                        if data.HealthBarBG then data.HealthBarBG.Visible = false end
                        if data.HealthBarFG then data.HealthBarFG.Visible = false end
                        if data.DistText then data.DistText.Visible = false end

                local distTextStr = string.format("%.0fm", distance)
                local baseScale = math.clamp(100 / (distance + 15), 0.35, 2.0)

                local fontSize = math.clamp(14 * finalScale, 5, 40)
                local headSize = math.clamp(16 * finalScale, 3, 45)

                        local half = size / 2
                        for x = -1, 1, 2 do
                            for y = -1, 1, 2 do
                                for z = -1, 1, 2 do
                                    local corner = cframe * Vector3.new(half.X * x, half.Y * y, half.Z * z)
                                    local screenPos, onScreen = Camera:WorldToViewportPoint(corner)
                                    table.insert(points, Vector2.new(screenPos.X, screenPos.Y))
                                end
                            end
                        end

                        local minX, minY = math.huge, math.huge
                        local maxX, maxY = -math.huge, -math.huge
                        for _, pt in ipairs(points) do
                            minX = math.min(minX, pt.X)
                            minY = math.min(minY, pt.Y)
                            maxX = math.max(maxX, pt.X)
                            maxY = math.max(maxY, pt.Y)
                        end

                        local boxW = maxX - minX
                        local boxH = maxY - minY
                        if boxW >= 1 and boxH >= 1 then
                            updateLine(data.Box.Top, Vector2.new(minX, minY), Vector2.new(maxX, minY), color, thick)
                            updateLine(data.Box.Bottom, Vector2.new(minX, maxY), Vector2.new(maxX, maxY), color, thick)
                            updateLine(data.Box.Left, Vector2.new(minX, minY), Vector2.new(minX, maxY), color, thick)
                            updateLine(data.Box.Right, Vector2.new(maxX, minY), Vector2.new(maxX, maxY), color, thick)
                        end
                    end
                else
                    for _, line in pairs(data.Box) do line.Visible = false end
                end

                if settings.espHead then
                    data.HeadCircle.Size = UDim2.new(0, headSize, 0, headSize)
                    data.HeadCircle.Position = UDim2.new(0, headScreen.X - headSize/2, 0, headScreen.Y - headSize/2)
                    data.HeadCircle.ImageColor3 = color
                    data.HeadCircle.ImageTransparency = 0.2
                    data.HeadCircle.Visible = true
                else
                    data.HeadCircle.Visible = false
                end

                if settings.espName then
                    data.Name.Text = plr.Name
                    data.Name.Position = UDim2.new(0, headScreen.X, 0, headScreen.Y - 50 * finalScale)
                    data.Name.TextColor3 = color
                    data.Name.TextSize = math.floor(fontSize)
                    data.Name.Visible = true
                else
                    data.Name.Visible = false
                end

                if hum then

                    local maxHp = hum.MaxHealth

                    end
                    if maxHp <= 0 or maxHp == nil or maxHp == 0 then
                        maxHp = plr:GetAttribute("MaxHealth") or plr:GetAttribute("MaxHP") or 100
                    end

                    if ls then

                        if lsHealth and lsHealth.Value > 0 then
                            hp = lsHealth.Value
                        end
                        local lsMaxHealth = ls:FindFirstChild("MaxHealth") or ls:FindFirstChild("MaxHP")
                        if lsMaxHealth and lsMaxHealth.Value > 0 then
                            maxHp = lsMaxHealth.Value
                        end
                    end
                    
                    if hp <= 0 or hp == nil then hp = 100 end
                    if maxHp <= 0 or maxHp == nil then maxHp = 100 end
                    
                    data.LastHealth = hp
                    data.LastMaxHealth = maxHp
                    
                    local healthPct = math.clamp(hp / maxHp, 0, 1)
                    local healthColor = Color3.fromRGB(255 * (1 - healthPct), 255 * healthPct, 0)
                    
                    if settings.espHealth then
                        data.HealthText.Text = string.format("%.0f/%.0f", hp, maxHp)
                        data.HealthText.Position = UDim2.new(0, headScreen.X, 0, headScreen.Y - 35 * finalScale)
                        data.HealthText.TextColor3 = healthColor
                        data.HealthText.TextSize = math.floor(fontSize * 0.8)
                        data.HealthText.Visible = true
                        data.HealthBarBG.Visible = false
                        data.HealthBarFG.Visible = false
                    elseif settings.espHealthBar then
                        local barWidth = math.clamp(35 * finalScale, 8, 70)
                        local barHeight = math.clamp(4 * finalScale, 2, 10)
                        local barX = headScreen.X - barWidth/2
                        local barY = headScreen.Y - 35 * finalScale
                        updateLine(data.HealthBarBG, Vector2.new(barX, barY), Vector2.new(barX + barWidth, barY), Color3.fromRGB(30, 30, 30), barHeight + 1)
                        updateLine(data.HealthBarFG, Vector2.new(barX, barY), Vector2.new(barX + barWidth * healthPct, barY), healthColor, barHeight)
                        data.HealthBarBG.Visible = true
                        data.HealthBarFG.Visible = true
                        data.HealthText.Visible = false
                    else
                        data.HealthText.Visible = false
                        data.HealthBarBG.Visible = false
                        data.HealthBarFG.Visible = false
                    end
                else
                    data.HealthText.Visible = false
                    data.HealthBarBG.Visible = false
                    data.HealthBarFG.Visible = false
                end

                if settings.espDistance then
                    data.DistText.Text = distTextStr
                    data.DistText.Position = UDim2.new(0, headScreen.X, 0, headScreen.Y - 18 * finalScale)
                    data.DistText.TextColor3 = Color3.fromRGB(180, 180, 180)
                    data.DistText.TextSize = math.floor(fontSize * 0.7)
                    data.DistText.Visible = true
                else
                    data.DistText.Visible = false
                end

                if settings.espWeapon then
                    local weapon = getPlayerWeapon(char)
                    if weapon then
                        local label = Instance.new("TextLabel")
                        label.BackgroundTransparency = 1
                        label.Text = weapon
                        label.TextColor3 = Color3.fromRGB(255, 200, 0)
                        label.Font = Enum.Font.Code
                        label.TextSize = math.floor(fontSize * 0.8)
                        label.TextStrokeTransparency = 0.2
                        label.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
                        label.Position = UDim2.new(0, headScreen.X, 0, headScreen.Y + 20 * finalScale)
                        label.Size = UDim2.new(0, 200, 0, 20)
                        label.AnchorPoint = Vector2.new(0.5, 0)
                        label.Parent = espScreenGui
                        table.insert(WeaponLabels, label)
                    end
                end

                if settings.espTracer then
                    local screenPos = Camera:WorldToViewportPoint(root.Position)
                    if screenPos then
                        local line = createLine()
                        local viewport = Camera.ViewportSize
                        local fromX = viewport.X / 2
                        local fromY = 0
                        
                        if settings.tracerPosition == "Middle" then
                            fromY = viewport.Y / 2
                        elseif settings.tracerPosition == "Bottom" then
                            fromY = viewport.Y
                        end
                        
                        updateLine(line, Vector2.new(fromX, fromY), Vector2.new(screenPos.X, screenPos.Y), color, 1.5)
                        table.insert(TracerLines, line)
                    end
                end
            end
            
            if settings.showEnemyCount then
                countLabel.Text = "P:" .. playerCount
                countFrame.Visible = true
            else
                countFrame.Visible = false
            end
        end)
    end)

    local function getAimbotCenter()
        if settings.aimbotMode == "Mouse" then
            local mousePos = UserInputService:GetMouseLocation()
            return Vector2.new(mousePos.X, mousePos.Y)
        else
            return Vector2.new(Camera.ViewportSize.X/2, Camera.ViewportSize.Y/2)
        end
    end

    local function getClosestTarget()
        local center = getAimbotCenter()
        local best, closest = nil, settings.aimbotFOV
        local targetPartName = settings.aimbotPart
        local maxDist = settings.aimbotMaxDist
        local cameraPos = Camera.CFrame.Position
        
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr == LocalPlayer then continue end
            
            if settings.aimbotTeamCheck then
                local sameTeam = isSameTeam(plr)
                if sameTeam then continue end
            end
            
            local char = plr.Character
            if not char then continue end
            local hum = char:FindFirstChildOfClass("Humanoid")
            if not hum or hum.Health <= 0 then continue end
            
            local targetPart = char:FindFirstChild(targetPartName) or char:FindFirstChild("HumanoidRootPart")
            if not targetPart then continue end
            
            local dist = (cameraPos - targetPart.Position).Magnitude
            if dist > maxDist then continue end
            
            if settings.aimbotVisibleCheck then
                local origin = Camera.CFrame.Position
                local dir = targetPart.Position - origin
                local params = RaycastParams.new()
                params.FilterDescendantsInstances = {LocalPlayer.Character}
                params.FilterType = Enum.RaycastFilterType.Exclude
                params.IgnoreWater = true
                local hit = Workspace:Raycast(origin, dir, params)
                if hit and not hit.Instance:IsDescendantOf(char) then continue end
            end
            
            local pos = targetPart.Position
            local screen, onScreen = Camera:WorldToViewportPoint(pos)
            if not onScreen then continue end
            
            local dist2d = (Vector2.new(screen.X, screen.Y) - center).Magnitude
            if dist2d < closest then
                closest = dist2d
                best = targetPart
            end
        end
        
        return best
    end

    RunService.RenderStepped:Connect(function()
        local shouldAim = false
        
        if settings.aimbotTrigger == "Auto" then
            shouldAim = settings.aimbotEnabled and aimbotAutoActive
        else
            shouldAim = settings.aimbotEnabled and isHoldingRMB
        end
        
        if not shouldAim then return end
        
        local target = getClosestTarget()
        if not target then return end
        
        local lookAt = CFrame.lookAt(Camera.CFrame.Position, target.Position)
        local alpha = math.clamp(settings.aimbotSmooth, 0.01, 1)
        if alpha >= 1 then
            Camera.CFrame = lookAt
        else
            Camera.CFrame = Camera.CFrame:Lerp(lookAt, alpha)
        end
    end)
